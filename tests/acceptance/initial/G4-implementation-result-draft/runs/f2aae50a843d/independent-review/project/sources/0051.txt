1: """审计日志记录模块。
2: 
3: 保留操作者、原状态、新状态和操作身份。
4: 重复撤回（幂等）不新增审计记录。
5: """
6: 
7: from __future__ import annotations
8: 
9: from dataclasses import dataclass, field
10: from typing import List
11: 
12: from .domain import ApplicationStatus
13: 
14: 
15: @dataclass
16: class AuditEntry:
17:     """一条审计记录。
18: 
19:     Attributes:
20:         operator_member_id: 执行操作的成员身份。
21:         application_id: 被操作的申请身份。
22:         previous_status: 操作前的状态。
23:         new_status: 操作后的状态。
24:         action: 操作类型（如"撤回"）。
25:     """
26: 
27:     operator_member_id: str
28:     application_id: str
29:     previous_status: ApplicationStatus
30:     new_status: ApplicationStatus
31:     action: str
32: 
33: 
34: class AuditLog:
35:     """审计日志存储。
36: 
37:     简单的内存存储，保存所有审计记录。
38:     """
39: 
40:     def __init__(self) -> None:
41:         self._entries: List[AuditEntry] = []
42: 
43:     def record(self, entry: AuditEntry) -> None:
44:         """记录一条审计条目。"""
45:         self._entries.append(entry)
46: 
47:     def entries_for(self, application_id: str) -> List[AuditEntry]:
48:         """返回指定申请的全部审计记录。"""
49:         return [e for e in self._entries if e.application_id == application_id]
50: 
51:     @property
52:     def all_entries(self) -> List[AuditEntry]:
53:         """返回全部审计记录（只读副本）。"""
54:         return list(self._entries)
