1: """预约申请领域模型。
2: 
3: 定义申请实体、生命周期状态、状态转换规则和撤回业务逻辑。
4: 身份信任来自可信宿主会话，不信任传入角色或所有者字段。
5: """
6: 
7: from __future__ import annotations
8: 
9: import enum
10: import uuid
11: from dataclasses import dataclass, field
12: from typing import Optional
13: 
14: 
15: class ApplicationStatus(enum.Enum):
16:     """预约申请生命周期状态。"""
17: 
18:     PENDING = "待审核"
19:     APPROVED = "已批准"
20:     REJECTED = "已拒绝"
21:     WITHDRAWN = "已撤回"
22: 
23: 
24: class WithdrawalResult(enum.Enum):
25:     """撤回操作结果。"""
26: 
27:     SUCCESS = "success"
28:     ALREADY_WITHDRAWN = "already_withdrawn"
29:     NOT_OWNER = "not_owner"
30:     INVALID_STATUS = "invalid_status"
31:     CONFLICT = "conflict"
32: 
33: 
34: @dataclass
35: class Application:
36:     """预约申请实体。
37: 
38:     每份申请有独立身份和版本号，用于乐观并发控制。
39:     """
40: 
41:     application_id: str
42:     owner_member_id: str
43:     status: ApplicationStatus = ApplicationStatus.PENDING
44:     version: int = 1
45: 
46:     @staticmethod
47:     def create(owner_member_id: str) -> "Application":
48:         """创建一份新的待审核申请。"""
49:         return Application(
50:             application_id=str(uuid.uuid4()),
51:             owner_member_id=owner_member_id,
52:             status=ApplicationStatus.PENDING,
53:             version=1,
54:         )
55: 
56:     def can_withdraw(self) -> bool:
57:         """检查当前状态是否允许撤回。"""
58:         return self.status == ApplicationStatus.PENDING
59: 
60:     def is_already_withdrawn(self) -> bool:
61:         """检查是否已处于已撤回状态。"""
62:         return self.status == ApplicationStatus.WITHDRAWN
63: 
64:     def withdraw(self, expected_version: int) -> bool:
65:         """执行撤回状态转换，使用乐观并发控制。
66: 
67:         Args:
68:             expected_version: 操作者期望的版本号，必须与当前版本匹配。
69: 
70:         Returns:
71:             True 表示成功转换，False 表示版本冲突（并发安全）。
72:         """
73:         if self.version != expected_version:
74:             return False
75:         self.status = ApplicationStatus.WITHDRAWN
76:         self.version += 1
77:         return True
78: 
79:     def approve(self, expected_version: int) -> bool:
80:         """执行审批状态转换，使用乐观并发控制。
81: 
82:         Args:
83:             expected_version: 操作者期望的版本号。
84: 
85:         Returns:
86:             True 表示成功转换，False 表示版本冲突。
87:         """
88:         if self.version != expected_version:
89:             return False
90:         self.status = ApplicationStatus.APPROVED
91:         self.version += 1
92:         return True
