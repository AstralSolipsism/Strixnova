107: class TestRepeatedWithdrawalIdempotent:
108:     """DIREX-A000000000000003: 重复撤回幂等。
109: 
110:     Given: 申请已处于已撤回状态
111:     When:  成员再次撤回该申请
112:     Then:  返回已撤回结果，不新增审计记录
113:     """
114: 
115:     def test_repeated_withdrawal_idempotent(self, service: MemberService) -> None:
116:         # Given: 先成功撤回一次
117:         app = _make_pending_application(service, owner_id="member-001")
118:         first_response = service.withdraw(
119:             authenticated_member_id="member-001",
120:             application_id=app.application_id,
121:         )
122:         assert first_response.result == WithdrawalResult.SUCCESS
123:         assert app.status == ApplicationStatus.WITHDRAWN
124: 
125:         audit_count_after_first = len(
126:             service.audit_log.entries_for(app.application_id)
127:         )
128:         assert audit_count_after_first == 1
129: 
130:         # When: 再次撤回
131:         second_response = service.withdraw(
132:             authenticated_member_id="member-001",
133:             application_id=app.application_id,
134:         )
135: 
136:         # Then: 返回已撤回结果
137:         assert second_response.result == WithdrawalResult.ALREADY_WITHDRAWN
138:         assert second_response.current_status == ApplicationStatus.WITHDRAWN
139: 
140:         # Then: 不新增审计记录
141:         audit_count_after_second = len(
142:             service.audit_log.entries_for(app.application_id)
143:         )
144:         assert audit_count_after_second == audit_count_after_first

147: class TestConcurrentWithdrawalAndApproval:
148:     """DIREX-A000000000000004: 并发撤回与审批。
149: 
150:     Given: 申请处于待审核状态
151:     When:  成员撤回和负责人审批同时发生
152:     Then:  只有一个操作成功，另一个被拒绝
153:     """
154: 
155:     def test_concurrent_withdrawal_and_approval(
156:         self, service: MemberService
157:     ) -> None:
158:         # Given
159:         app = _make_pending_application(service, owner_id="member-001")
160:         assert app.status == ApplicationStatus.PENDING
161: 
162:         # 模拟并发：两方同时读取当前版本
163:         version_seen_by_member = app.version
164:         version_seen_by_approver = app.version
165: 
166:         # When: 成员先执行撤回（使用读到的版本）
167:         withdraw_succeeded = app.withdraw(version_seen_by_member)
168: 
169:         # When: 审批者尝试使用旧版本审批
170:         approve_succeeded = app.approve(version_seen_by_approver)
171: 
172:         # Then: 只有一个操作成功
173:         assert withdraw_succeeded is True
174:         assert approve_succeeded is False
175:         assert app.status == ApplicationStatus.WITHDRAWN
176: 
177:         # 反方向场景验证：如果审批者先执行
178:         app2 = _make_pending_application(service, owner_id="member-001")
179:         version_a = app2.version
180:         version_b = app2.version
181: 
182:         # 审批者先执行
183:         approve_first = app2.approve(version_a)
184:         # 成员再撤回
185:         withdraw_second = app2.withdraw(version_b)
186: 
187:         assert approve_first is True
188:         assert withdraw_second is False
189:         assert app2.status == ApplicationStatus.APPROVED
